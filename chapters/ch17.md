### 비용 모니터링 스크립트

```python
#!/usr/bin/env python3
# monitor_claude_cost.py
import json
import subprocess
from datetime import datetime, timedelta

def get_anthropic_usage(start_date: str, end_date: str) -> dict:
    """Anthropic API 사용량 조회"""
    # Anthropic Console API 사용
    import requests
    headers = {"x-api-key": os.environ["ANTHROPIC_ADMIN_KEY"]}
    response = requests.get(
        f"https://api.anthropic.com/v1/organizations/usage",
        params={"start_date": start_date, "end_date": end_date},
        headers=headers
    )
    return response.json()

def analyze_cost_trend():
    """최근 7일 비용 트렌드 분석"""
    today = datetime.now()
    week_ago = today - timedelta(days=7)

    usage = get_anthropic_usage(
        week_ago.strftime("%Y-%m-%d"),
        today.strftime("%Y-%m-%d")
    )

    total_cost = sum(day["total_cost_usd"] for day in usage["daily"])
    print(f"최근 7일 총 비용: ${total_cost:.2f}")

    # 일별 트렌드
    for day in usage["daily"]:
        print(f"  {day['date']}: ${day['total_cost_usd']:.2f}")

    # 월간 예측
    daily_avg = total_cost / 7
    monthly_est = daily_avg * 30
    print(f"\n월간 예상 비용: ${monthly_est:.2f}")

if __name__ == "__main__":
    analyze_cost_trend()
```

### 권한 감사 로그

```python
# ~/.claude/hooks/audit-log.py
import sys
import json
import os
from datetime import datetime

LOG_FILE = os.path.expanduser("~/.claude/audit.log")

def main():
    data = json.load(sys.stdin)
    tool_name = data.get("tool_name", "")
    tool_input = data.get("tool_input", {})
    session_id = data.get("session_id", "")[:8]

    log_entry = {
        "timestamp": datetime.now().isoformat(),
        "session": session_id,
        "tool": tool_name,
        "input_summary": str(tool_input)[:200]  # 처음 200자만
    }

    with open(LOG_FILE, "a") as f:
        f.write(json.dumps(log_entry, ensure_ascii=False) + "\n")

    # 통과
    print(json.dumps({"decision": "allow"}))

if __name__ == "__main__":
    main()
```

```python
# ~/.claude/hooks/redact-secrets.py
import sys
import json
import re

# 민감한 패턴
SENSITIVE_PATTERNS = [
    (r'sk-ant-api\w+', '[ANTHROPIC_KEY]'),
    (r'sk_live_\w+', '[STRIPE_LIVE_KEY]'),
    (r'sk_test_\w+', '[STRIPE_TEST_KEY]'),
    (r'ghp_\w+', '[GITHUB_TOKEN]'),
    (r'password\s*=\s*["\']([^"\']+)["\']', 'password="[REDACTED]"'),
    (r'\b\d{4}[-\s]\d{4}[-\s]\d{4}[-\s]\d{4}\b', '[CARD_NUMBER]'),
]

def redact_output(text: str) -> str:
    for pattern, replacement in SENSITIVE_PATTERNS:
        text = re.sub(pattern, replacement, text, flags=re.IGNORECASE)
    return text

def main():
    data = json.load(sys.stdin)
    # 출력 내용에서 민감한 정보 마스킹
    if "tool_response" in data:
        response_str = json.dumps(data["tool_response"])
        redacted = redact_output(response_str)
        if redacted != response_str:
            print(f"⚠️ 민감한 정보가 감지되어 마스킹되었습니다", file=sys.stderr)

if __name__ == "__main__":
    main()
```

### 개인정보 처리 가이드라인

```
Claude Code로 작업할 때 개인정보 보호:

1. 실제 사용자 데이터 사용 금지
   → 테스트 데이터 생성기(faker) 사용

2. 로그에 개인정보 포함 금지
   → user_id만 로깅, 이름/이메일은 마스킹

3. AI에 개인정보 전송 금지
   → 에러 분석 시 실제 데이터 대신 샘플 데이터 사용

4. 프로덕션 DB 접근 금지
   → 개발 DB 또는 익명화된 스냅샷 사용

예시:
❌ "이 사용자 데이터를 분석해줘: {name: '홍길동', phone: '010-1234-5678'}"
✅ "이 형식의 데이터를 분석해줘: {name: 'USER_NAME', phone: 'PHONE_NUMBER'}"
```

## 핵심 요약

1. `/cost` 명령과 Anthropic Console로 **비용을 실시간 모니터링**합니다
2. `.claudeignore`와 컨텍스트 최소화로 **불필요한 비용을 줄입니다**
3. `settings.json`의 허용/차단 목록으로 **최소 권한 원칙**을 적용합니다
4. 보안 훅으로 **민감한 정보 유출**을 자동으로 방지합니다
5. 개인정보는 절대 Claude Code에 전송하지 않고, **익명화된 데이터**만 사용합니다

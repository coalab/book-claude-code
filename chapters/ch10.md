## 10.1 훅 시스템 개요: PreToolUse, PostToolUse, Stop 이벤트

Claude Code의 훅 시스템은 AI가 특정 도구를 사용하거나 작업을 완료할 때 사용자 정의 스크립트를 실행합니다. 이를 통해 Claude Code의 동작을 세밀하게 제어할 수 있습니다.

### 이벤트 종류

```
훅 이벤트 흐름:

사용자 요청
    ↓
[PreToolUse 훅] ← 도구 실행 전 개입 가능
    ↓
Claude Code 도구 실행
    ↓
[PostToolUse 훅] ← 도구 실행 후 처리
    ↓
응답 생성
    ↓
[Stop 훅] ← 세션/작업 완료 시
```

| 이벤트 | 발생 시점 | 가능한 행동 |
|--------|----------|-----------|
| `PreToolUse` | 도구 실행 직전 | 차단(block), 수정, 로깅 |
| `PostToolUse` | 도구 실행 직후 | 후처리, 알림, 검증 |
| `Notification` | 알림 발생 시 | 외부 알림 전송 |
| `Stop` | 에이전트 완료 시 | 정리 작업, 보고 |

### 훅이 받는 데이터

훅 스크립트는 JSON 형태로 이벤트 데이터를 받습니다:

```json
// PreToolUse 이벤트 데이터 (stdin으로 전달)
{
  "session_id": "abc123",
  "tool_name": "Bash",
  "tool_input": {
    "command": "rm -rf ./temp"
  }
}
```

```json
// PostToolUse 이벤트 데이터
{
  "session_id": "abc123",
  "tool_name": "Write",
  "tool_input": {
    "file_path": "src/utils/helper.ts",
    "content": "..."
  },
  "tool_response": {
    "success": true
  }
}
```

## 10.2 훅 설정 파일 작성법 (settings.json)

훅은 `.claude/settings.json` 파일에 설정합니다.

### 기본 구조

```json
{
  "hooks": {
    "PreToolUse": [
      {
        "matcher": "Bash",
        "hooks": [
          {
            "type": "command",
            "command": "python3 ~/.claude/hooks/pre-bash.py"
          }
        ]
      }
    ],
    "PostToolUse": [
      {
        "matcher": "Write",
        "hooks": [
          {
            "type": "command",
            "command": "bash ~/.claude/hooks/post-write.sh"
          }
        ]
      }
    ],
    "Stop": [
      {
        "hooks": [
          {
            "type": "command",
            "command": "python3 ~/.claude/hooks/on-stop.py"
          }
        ]
      }
    ]
  }
}
```

### matcher 패턴

```json
// 특정 도구에만 훅 적용
"matcher": "Bash"

// 도구 + 입력 패턴으로 매칭
"matcher": "Bash(npm *)"

// 파일 경로 패턴
"matcher": "Write(*.ts)"

// 모든 도구
"matcher": ""  // 또는 생략
```

### 예제 1: TypeScript 파일 자동 포맷팅

  "hooks": {

### 예제 2: 파이썬 훅 스크립트 (린팅)

```python
# ~/.claude/hooks/post-write.py
import sys
import json
import subprocess
import os

def main():
    # stdin에서 이벤트 데이터 읽기
    data = json.load(sys.stdin)

    tool_name = data.get("tool_name", "")
    tool_input = data.get("tool_input", {})
    file_path = tool_input.get("file_path", "")

    # Python 파일인 경우 flake8 실행
    if file_path.endswith(".py"):
        result = subprocess.run(
            ["flake8", "--max-line-length=100", file_path],
            capture_output=True,
            text=True
        )
        if result.returncode != 0:
            # 경고 출력 (차단하지 않고 알림만)
            print(f"⚠️ 린팅 경고:\n{result.stdout}", file=sys.stderr)

    # TypeScript 파일인 경우 타입 체크
    if file_path.endswith((".ts", ".tsx")):
        result = subprocess.run(
            ["npx", "tsc", "--noEmit", "--skipLibCheck"],
            capture_output=True,
            text=True,
            cwd=os.path.dirname(file_path)
        )
        if result.returncode != 0:
            print(f"⚠️ TypeScript 오류:\n{result.stdout}", file=sys.stderr)

if __name__ == "__main__":
    main()
```

### 예제 3: Slack 알림 전송

```python
# ~/.claude/hooks/notify-slack.py
import sys
import json
import requests
import os
from datetime import datetime

SLACK_WEBHOOK = os.environ.get("SLACK_WEBHOOK_URL")

def send_slack_message(message: str):
    if not SLACK_WEBHOOK:
        return
    requests.post(SLACK_WEBHOOK, json={"text": message}, timeout=5)

def main():
    data = json.load(sys.stdin)
    session_id = data.get("session_id", "unknown")[:8]

    # 세션 완료 시 요약 알림
    message = (
        f"🤖 Claude Code 작업 완료\n"
        f"세션: {session_id}\n"
        f"시간: {datetime.now().strftime('%H:%M:%S')}\n"
        f"프로젝트: {os.getcwd().split('/')[-1]}"
    )
    send_slack_message(message)

if __name__ == "__main__":
    main()
```

## 10.4 보안 훅 구현: 민감한 파일 보호와 위험 명령 차단

### 위험 명령어 차단

```python
# ~/.claude/hooks/security-pre-bash.py
import sys
import json
import re

# 절대 실행하면 안 되는 패턴
BLOCKED_PATTERNS = [
    r"rm\s+-rf\s+/",           # 루트 디렉토리 삭제
    r"rm\s+-rf\s+~",           # 홈 디렉토리 삭제
    r">\s*/etc/",              # /etc 파일 덮어쓰기
    r"curl.*\|\s*bash",        # 원격 스크립트 실행
    r"wget.*\|\s*sh",          # 원격 스크립트 실행
    r"chmod\s+777",            # 너무 넓은 권한
]

# 경고가 필요한 패턴
WARNING_PATTERNS = [
    r"sudo\s+",                # sudo 사용
    r"git\s+push\s+.*--force", # 강제 푸시
    r"DROP\s+TABLE",           # DB 테이블 삭제
    r"DELETE\s+FROM.*WHERE",   # 대량 삭제
]

def main():
    data = json.load(sys.stdin)
    command = data.get("tool_input", {}).get("command", "")

    # 차단 패턴 검사
    for pattern in BLOCKED_PATTERNS:
        if re.search(pattern, command, re.IGNORECASE):
            print(json.dumps({
                "decision": "block",
                "reason": f"보안 정책 위반: {pattern} 패턴 감지\n명령어: {command}"
            }))
            return

    # 경고 패턴 검사 (차단하지 않고 로그만)
    for pattern in WARNING_PATTERNS:
        if re.search(pattern, command, re.IGNORECASE):
            print(f"⚠️ 주의 명령어 실행: {command}", file=sys.stderr)

    # 통과
    print(json.dumps({"decision": "allow"}))

if __name__ == "__main__":
    main()
```

### 민감한 파일 보호

```python
# ~/.claude/hooks/protect-sensitive-files.py
import sys
import json
import os

# 수정 금지 파일/패턴
PROTECTED_PATHS = [
    ".env",
    ".env.production",
    ".env.local",
    "secrets.json",
    "credentials.json",
    "private.key",
    "*.pem",
]

def is_protected(file_path: str) -> bool:
    file_name = os.path.basename(file_path)
    for pattern in PROTECTED_PATHS:
        if pattern.startswith("*"):
            if file_name.endswith(pattern[1:]):
                return True
        elif file_name == pattern:
            return True
    return False

def main():
    data = json.load(sys.stdin)
    tool_name = data.get("tool_name", "")
    file_path = data.get("tool_input", {}).get("file_path", "")

    if tool_name in ("Write", "Edit") and file_path:
        if is_protected(file_path):
            print(json.dumps({
                "decision": "block",
                "reason": f"보호된 파일입니다: {file_path}\n이 파일을 수정하려면 직접 편집기를 사용하세요."
            }))
            return

    print(json.dumps({"decision": "allow"}))

if __name__ == "__main__":
    main()
```

> **💡 팁:** 훅은 팀 전체에 공유할 수 있습니다. `.claude/hooks/` 폴더를 git에 포함시키고 `settings.json`에서 참조하면, 모든 팀원이 동일한 보안 정책을 자동으로 적용받습니다.

## 핵심 요약

1. **훅 시스템**은 PreToolUse, PostToolUse, Stop 이벤트에서 스크립트를 실행합니다
2. **settings.json**에서 이벤트, matcher 패턴, 실행 명령을 설정합니다
3. PostToolUse 훅으로 **자동 포맷팅과 린팅**을 구현합니다
4. PreToolUse 훅으로 **위험한 명령어를 차단**합니다
5. 훅을 **git에 포함**하여 팀 전체에 일관된 정책을 적용합니다

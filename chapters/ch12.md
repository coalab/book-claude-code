### API 엔드포인트 설계 먼저

### Task Service 구현

```typescript
// packages/api/src/services/task.service.ts
import { PrismaClient, Task, TaskStatus, Priority } from "@prisma/client";
import { injectable, inject } from "tsyringe";
import { TaskRepository } from "../repositories/task.repository";

export interface CreateTaskDto {
  title: string;
  description?: string;
  priority?: Priority;
  dueDate?: Date;
  assigneeId?: string;
  workspaceId: string;
}

export interface UpdateTaskDto {
  title?: string;
  description?: string;
  priority?: Priority;
  dueDate?: Date;
  assigneeId?: string | null;
}

@injectable()
export class TaskService {
  constructor(
    @inject("TaskRepository") private taskRepo: TaskRepository,
    @inject("PrismaClient") private prisma: PrismaClient
  ) {}

  async createTask(data: CreateTaskDto, userId: string): Promise<Task> {
    // 워크스페이스 멤버십 확인
    const membership = await this.prisma.workspaceMember.findUnique({
      where: {
        userId_workspaceId: { userId, workspaceId: data.workspaceId }
      }
    });

    if (!membership) {
      throw new Error("워크스페이스 접근 권한이 없습니다");
    }

    return this.taskRepo.create(data);
  }

  async getTasksByWorkspace(
    workspaceId: string,
    userId: string,
    filters?: { status?: TaskStatus; assigneeId?: string }
  ): Promise<Task[]> {
    // 권한 확인
    await this.assertWorkspaceMember(workspaceId, userId);
    return this.taskRepo.findByWorkspace(workspaceId, filters);
  }

  async updateTaskStatus(
    taskId: string,
    status: TaskStatus,
    userId: string
  ): Promise<Task> {
    const task = await this.taskRepo.findById(taskId);
    if (!task) throw new Error("태스크를 찾을 수 없습니다");

    await this.assertWorkspaceMember(task.workspaceId, userId);
    return this.taskRepo.update(taskId, { status });
  }

  private async assertWorkspaceMember(
    workspaceId: string,
    userId: string
  ): Promise<void> {
    const member = await this.prisma.workspaceMember.findUnique({
      where: { userId_workspaceId: { userId, workspaceId } }
    });
    if (!member) throw new Error("접근 권한이 없습니다");
  }
}
```

## 12.3 프론트엔드 컴포넌트 개발 (React + Tailwind CSS)

### 태스크 보드 컴포넌트

```tsx
// packages/web/src/components/TaskBoard.tsx
import React, { useState } from "react";
import { useTasks } from "../hooks/useTasks";
import { TaskCard } from "./TaskCard";
import { TaskStatus } from "../types/api";

const COLUMNS = [
  { id: "TODO" as TaskStatus, label: "할 일", color: "bg-gray-100" },
  { id: "IN_PROGRESS" as TaskStatus, label: "진행 중", color: "bg-blue-100" },
  { id: "DONE" as TaskStatus, label: "완료", color: "bg-green-100" },
];

interface TaskBoardProps {
  workspaceId: string;
}

export function TaskBoard({ workspaceId }: TaskBoardProps) {
  const { tasks, isLoading, updateTaskStatus } = useTasks(workspaceId);

  if (isLoading) {
    return (
      <div className="flex items-center justify-center h-64">
        <div className="animate-spin rounded-full h-8 w-8 border-b-2 border-blue-600" />
      </div>
    );
  }

  return (
    <div className="flex gap-4 p-6 overflow-x-auto">
      {COLUMNS.map((column) => {
        const columnTasks = tasks.filter((t) => t.status === column.id);

        return (
          <div key={column.id} className="flex-shrink-0 w-72">
            <div className={`rounded-lg p-3 ${column.color}`}>
              <div className="flex items-center justify-between mb-3">
                <h3 className="font-semibold text-gray-700">{column.label}</h3>
                <span className="bg-white text-gray-600 text-xs font-medium px-2 py-1 rounded-full">
                  {columnTasks.length}
                </span>
              </div>

              <div className="space-y-2">
                {columnTasks.map((task) => (
                  <TaskCard
                    key={task.id}
                    task={task}
                    onStatusChange={(status) =>
                      updateTaskStatus(task.id, status)
                    }
                  />
                ))}
              </div>
            </div>
          </div>
        );
      })}
    </div>
  );
}
```

## 12.4 통합 테스트 작성과 배포 준비 (Docker, CI/CD)

### 통합 테스트

```typescript
// packages/api/tests/integration/tasks.test.ts
import request from "supertest";
import { app } from "../../src/app";
import { prisma } from "../../src/config/database";

describe("Task API 통합 테스트", () => {
  let authToken: string;
  let workspaceId: string;

  beforeAll(async () => {
    // 테스트 사용자 생성 및 로그인
    await request(app).post("/auth/register").send({
      email: "test@example.com",
      name: "테스터",
      password: "SecurePass123!",
    });

    const loginRes = await request(app).post("/auth/login").send({
      email: "test@example.com",
      password: "SecurePass123!",
    });
    authToken = loginRes.body.accessToken;

    // 테스트 워크스페이스 생성
    const wsRes = await request(app)
      .post("/workspaces")
      .set("Authorization", `Bearer ${authToken}`)
      .send({ name: "테스트 워크스페이스" });
    workspaceId = wsRes.body.id;
  });

  afterAll(async () => {
    await prisma.task.deleteMany();
    await prisma.workspaceMember.deleteMany();
    await prisma.workspace.deleteMany();
    await prisma.user.deleteMany();
    await prisma.$disconnect();
  });

  describe("POST /workspaces/:id/tasks", () => {
    it("태스크를 성공적으로 생성한다", async () => {
      const res = await request(app)
        .post(`/workspaces/${workspaceId}/tasks`)
        .set("Authorization", `Bearer ${authToken}`)
        .send({
          title: "첫 번째 태스크",
          priority: "HIGH",
          dueDate: new Date(Date.now() + 7 * 24 * 60 * 60 * 1000),
        });

      expect(res.status).toBe(201);
      expect(res.body.title).toBe("첫 번째 태스크");
      expect(res.body.status).toBe("TODO");
    });
  });
});
```
